{%- comment -%} Finds this page's translation (same `ref`, other language) and outputs it as either the language switch in the header or tags. Pages without a translation link to the other language's home page. Usage: {% include translation.html as="switch" %} or as="links" {%- endcomment -%} {%- assign here = site.data.i18n[page.lang] -%} {%- for pair in site.data.i18n -%} {%- if pair[0] != page.lang %}{% assign other_lang = pair[0] %}{% assign there = pair[1] %}{% endif -%} {%- endfor -%} {%- assign translation = nil -%} {%- if page.ref -%} {%- assign everything = site.pages | concat: site.documents -%} {%- assign translation = everything | where: 'ref', page.ref | where: 'lang', other_lang | first -%} {%- endif -%} {%- if translation %}{% assign target = translation.url %}{% else %}{% assign target = there.urls.home %}{% endif -%} {%- if include.as == 'links' -%} {%- if translation %} {%- endif %} {%- else -%} {%- comment %} The site's default language (site.lang) first, then the rest: ES | EN {% endcomment -%} {%- assign order = site.lang -%} {%- for pair in site.data.i18n -%} {%- if pair[0] != site.lang %}{% assign order = order | append: ',' | append: pair[0] %}{% endif -%} {%- endfor -%} {%- assign order = order | split: ',' -%}
{%- for lang in order %} {%- assign l = site.data.i18n[lang] %} {%- if lang == page.lang %} {{ l.code | upcase }} {%- else %} {{ l.code | upcase }}{{ l.switch_label }} {%- endif %} {%- endfor %}
{%- endif -%}
